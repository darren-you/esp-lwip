cmake_minimum_required(VERSION 3.16)
project(lwip_zero_window_test C)
set(CMAKE_C_STANDARD 11)
set(LWIP_SOURCE_DIR "${CMAKE_CURRENT_LIST_DIR}/../.." CACHE PATH "Explicit upstream lwIP source checkout")
if(NOT EXISTS "${LWIP_SOURCE_DIR}/src/Filelists.cmake")
    message(FATAL_ERROR "Set LWIP_SOURCE_DIR to the actual SDK lwIP checkout")
endif()
set(LWIP_DIR "${LWIP_SOURCE_DIR}")
include("${LWIP_DIR}/src/Filelists.cmake")
add_executable(lwip_loopback_test loopback_test.c ${lwipcore_SRCS} ${lwipcore4_SRCS})
target_include_directories(lwip_loopback_test PRIVATE . "${LWIP_DIR}/src/include")
target_compile_options(lwip_loopback_test PRIVATE -UNDEBUG)
include(CTest)
add_test(NAME lwip_zero_window_loopback COMMAND lwip_loopback_test)
add_test(NAME lwip_zero_window_wrap COMMAND lwip_loopback_test wrap)
set_tests_properties(lwip_zero_window_loopback lwip_zero_window_wrap PROPERTIES TIMEOUT 10)
